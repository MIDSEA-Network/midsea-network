people = FileAttachment("people.csv").csv({typed: false})

placeholderAvatar = new URL("/images/people/placeholder-avatar.svg", location.origin).href

// people.csv stores the SHA-256 of the lower-cased email, never the email
// itself: the CSV is published with the site. Gravatar looks up by that hash.
function gravatarUrl(hash) {
  // d=<placeholderAvatar> tells Gravatar to serve our own placeholder if the
  // email has no Gravatar registered, instead of Gravatar's generic icon.
  return `https://www.gravatar.com/avatar/${hash}?s=200&d=${encodeURIComponent(placeholderAvatar)}`;
}

function photoUrl(p) {
  if (p.photo && p.photo.trim()) return p.photo.trim();
  if (p.email_sha256 && p.email_sha256.trim()) return gravatarUrl(p.email_sha256.trim());
  return placeholderAvatar;
}

photoUrls = people.map(photoUrl)

html`<div class="people-grid">
  ${people.map((p, i) => html`
    <div class="person-card">
      <img src="${photoUrls[i]}" alt="${p.name}"
        onerror="this.onerror=null;this.src='${placeholderAvatar}';">
      <div class="person-card-body">
        <h3>${p.profile_url && p.profile_url.trim()
          ? html`<a href="${p.profile_url}">${p.name}</a>`
          : p.name}</h3>
        <p class="person-title">${p.title}</p>
        <p class="person-institution">${p.institution}</p>
      </div>
    </div>
  `)}
</div>`
